cmake_minimum_required(VERSION 3.20)
if(NOT TARGET lvgl)
    project(geektool_stocks_tests LANGUAGES C)
    set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../host/lv_conf.h" CACHE PATH "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
    enable_testing()
endif()
set(STOCK_MAIN ${CMAKE_CURRENT_SOURCE_DIR}/../../main)
set(STOCK_CJSON ${CMAKE_CURRENT_SOURCE_DIR}/../../managed_components/espressif__cjson/cJSON)
add_library(stocks_parser STATIC ${STOCK_MAIN}/stocks_data.c ${STOCK_CJSON}/cJSON.c)
target_include_directories(stocks_parser PUBLIC ${STOCK_MAIN} ${STOCK_CJSON})
target_compile_options(stocks_parser PRIVATE -Wall -Wextra -Werror)
foreach(kind IN ITEMS data network ui)
    add_executable(stocks_${kind}_tests ${kind}_tests.c)
    target_link_libraries(stocks_${kind}_tests PRIVATE stocks_parser m)
    target_compile_definitions(stocks_${kind}_tests PRIVATE STOCK_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
    target_compile_options(stocks_${kind}_tests PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
endforeach()
target_sources(stocks_data_tests PRIVATE ${STOCK_MAIN}/stocks_store.c)
target_include_directories(stocks_data_tests PRIVATE stubs)
target_sources(stocks_network_tests PRIVATE ${STOCK_MAIN}/stocks_fetch.c)
target_include_directories(stocks_network_tests PRIVATE ../answers/stubs)
target_sources(stocks_ui_tests PRIVATE ${STOCK_MAIN}/font_stocks.c ${STOCK_MAIN}/font_stocks_cjk.c ${STOCK_MAIN}/font_stocks_cjk_22.c
    ${STOCK_MAIN}/i18n.c ${STOCK_MAIN}/font_cn16.c ${STOCK_MAIN}/font_location_24.c)
target_link_libraries(stocks_ui_tests PRIVATE lvgl)
add_test(NAME stocks_symbols_quotes_and_persistence COMMAND stocks_data_tests)
add_test(NAME stocks_worker_cancellation_and_cleanup COMMAND stocks_network_tests)
add_test(NAME stocks_bilingual_ui_and_lifecycle COMMAND stocks_ui_tests)
